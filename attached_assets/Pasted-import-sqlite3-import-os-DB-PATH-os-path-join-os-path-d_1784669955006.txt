import sqlite3
import os

DB_PATH = os.path.join(os.path.dirname(__file__), 'vendashu.db')

def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db()
    c = conn.cursor()

    c.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL UNIQUE,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
    ''')

    c.execute('''
        CREATE TABLE IF NOT EXISTS shoes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER,
            shoe_type TEXT,
            construction TEXT,
            season TEXT,
            color TEXT,
            designer TEXT,
            subsection TEXT,
            bin_col TEXT,
            bin_row TEXT,
            bin_location TEXT,
            is_boots BOOLEAN DEFAULT 0,
            image_path TEXT,
            label_text TEXT,
            status TEXT DEFAULT 'stored',
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users(id)
        )
    ''')

    c.execute('''
        CREATE TABLE IF NOT EXISTS bins (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            subsection TEXT NOT NULL,
            bin_col TEXT NOT NULL,
            bin_row TEXT NOT NULL,
            bin_location TEXT NOT NULL,
            status TEXT DEFAULT 'available',
            shoe_id INTEGER,
            FOREIGN KEY (shoe_id) REFERENCES shoes(id)
        )
    ''')

    # Seed bins if empty
    c.execute('SELECT COUNT(*) FROM bins')
    if c.fetchone()[0] == 0:
        subsections = ['SS-A', 'SS-B', 'SS-C', 'SS-D', 'SS-E', 'SS-F']
        rows = ['R1', 'R2', 'R3', 'R4']
        locations = ['LF', 'R', 'RF']
        for ss in subsections:
            col = 'C' + str(subsections.index(ss) + 1)
            for row in rows:
                for loc in locations:
                    c.execute('''
                        INSERT INTO bins (subsection, bin_col, bin_row, bin_location)
                        VALUES (?, ?, ?, ?)
                    ''', (ss, col, row, loc))
                c.execute('''
                    INSERT INTO bins (subsection, bin_col, bin_row, bin_location)
                    VALUES (?, ?, ?, ?)
                ''', (ss, col, row, 'FB'))

    # Seed default users if empty
    c.execute('SELECT COUNT(*) FROM users')
    if c.fetchone()[0] == 0:
        default_users = ['David', 'Jessica', 'Angelica', 'Xavier', 'Emily', 'Justin']
        for user in default_users:
            c.execute('INSERT INTO users (name) VALUES (?)', (user,))

    conn.commit()
    conn.close()
    print("Database initialized.")

if __name__ == '__main__':
    init_db()
