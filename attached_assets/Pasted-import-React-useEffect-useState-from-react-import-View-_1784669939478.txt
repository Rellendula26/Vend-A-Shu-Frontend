function OnboardingScreen({ navigate }) {
  const [page, setPage] = useState(0);
  const slides = [
    { icon: '📶', title: 'Connect to Wifi', desc: 'Make sure you are connected to your Wifi Router to continue using this app.' },
    { icon: '👤', title: 'Create Users', desc: 'Connect to your VAS database and create users who can have access to the VAS system.' },
    { icon: '👟', title: 'Select User and Get Started!', desc: 'Select your username from the dropdown menu on the start page and get started.' },
    { icon: '📦', title: 'Know your Bins and Locations!', desc: 'Each bin can hold one, two or three items. Locations are labeled Left Front, Rear, and Right Front.' },
  ];
  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.onboardingCard}>
        <Text style={styles.onboardingIcon}>{slides[page].icon}</Text>
        <Text style={styles.onboardingTitle}>{slides[page].title}</Text>
        <Text style={styles.onboardingDesc}>{slides[page].desc}</Text>
        <View style={styles.dots}>
          {slides.map((_, i) => <View key={i} style={[styles.dot, i === page && styles.dotActive]} />)}
        </View>
      </View>
      <View style={styles.onboardingButtons}>
        {page < slides.length - 1 ? (
          <>

            <OrangeButton title="Get Started" onPress={() => navigate('connect')} />
            <GhostButton title="Back" onPress={() => setPage(page - 1)} />

function AddShoesScreen({ navigate, shoes, setShoes }) {
  const [step, setStep] = useState(1);
  const [shoeType, setShoeType] = useState('');
  const [construction, setConstruction] = useState('');
  const [season, setSeason] = useState('');
  const [color, setColor] = useState('');
  const [designer, setDesigner] = useState('');
  const [selectedBin, setSelectedBin] = useState(null);
  const [selectedVAS, setSelectedVAS] = useState('');
  const [vasOpen, setVasOpen] = useState(false);
  const [seasonOpen, setSeasonOpen] = useState(false);
  const seasons = ['Winter', 'Summer', 'Spring', 'Fall', 'All'];
  const isBoots = /boots?/i.test(shoeType);
  const binGrid = Array.from({ length: 60 }, (_, i) => ({ id: i, occupied: [2,5,11,18,23,30,35,42].includes(i) }));

  const handleReserveBin = () => {
    const newShoe = { id: Date.now().toString(), type: shoeType, construction, season, color, designer, bin: selectedVAS || 'Master Bottom', col: `C${Math.floor(selectedBin / 10) + 1}`, row: `R${(selectedBin % 10) + 1}`, location: isBoots ? 'Full Bin' : 'Right', emoji: isBoots ? '🥾' : '👟' };
    setShoes([...shoes, newShoe]);
    setStep(4);
  };

  if (step === 4) return (
    <SafeAreaView style={styles.container}>
      <AppHeader />
      <View style={styles.successContainer}>
        <View style={styles.successCircle}><Text style={styles.successCheck}>✓</Text></View>
        <Text style={styles.successTitle}>Shoes Added!</Text>
        <Text style={styles.successSub}>{selectedVAS} : {`C${Math.floor((selectedBin||0)/10)+1}`} : {`R${((selectedBin||0)%10)+1}`}</Text>
        <View style={{ marginTop: 40, width: '100%' }}><OrangeButton title="Go to Home" onPress={() => navigate('home')} /></View>
      </View>
    </SafeAreaView>
  );

  if (step === 3) return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="" onBack={() => setStep(2)} />
      <ScrollView contentContainerStyle={styles.scrollContent}>
        <Text style={styles.cursiveTitle}>Select location to place the shoes</Text>
        <TouchableOpacity style={styles.dropdown} onPress={() => setVasOpen(!vasOpen)}>
          <Text style={selectedVAS ? styles.dropdownSelected : styles.dropdownPlaceholder}>{selectedVAS || 'Select VAS'}</Text>
          <Text>▼</Text>
        </TouchableOpacity>
        {vasOpen && (
          <View style={styles.dropdownList}>
            {VAS_SYSTEMS.map(v => (
              <TouchableOpacity key={v} style={styles.dropdownItem} onPress={() => { setSelectedVAS(v); setVasOpen(false); }}>
                <Text style={styles.dropdownItemText}>{v}</Text>
              </TouchableOpacity>
            ))}
          </View>
        )}
        <View style={styles.binGrid}>
          {binGrid.map(bin => (
            <TouchableOpacity key={bin.id} style={[styles.binCell, bin.occupied && styles.binOccupied, selectedBin === bin.id && styles.binSelected]} onPress={() => !bin.occupied && setSelectedBin(bin.id)} />
          ))}
        </View>
        <View style={styles.helpBox}>
          <Text style={styles.helpTitle}>💡 How to Select the Storage Location?</Text>
          <Text style={styles.helpText}>The app has already auto selected the storage location for you, but you may change it by clicking on a different storage location.</Text>
        </View>
        <OrangeButton title="Use Selected Bin" onPress={handleReserveBin} disabled={!selectedBin || !selectedVAS} />
      </ScrollView>
    </SafeAreaView>
  );

  if (step === 2) return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="" onBack={() => setStep(1)} />
      <ScrollView contentContainerStyle={styles.scrollContent}>
        <Text style={styles.cursiveTitle}>Add Shoes or Other Items</Text>
        <View style={styles.uploadBox}>
          <Text style={styles.uploadIcon}>⬆️</Text>
          <Text style={styles.uploadText}>Click to upload image</Text>
          <Text style={styles.uploadSub}>Maximum file size 20MB</Text>
        </View>
        <TextInput style={styles.input} placeholder="Shoe Type" value={shoeType} onChangeText={setShoeType} />
        <TextInput style={styles.input} placeholder="Construction of Shoes" value={construction} onChangeText={setConstruction} />
        <TouchableOpacity style={styles.dropdown} onPress={() => setSeasonOpen(!seasonOpen)}>
          <Text style={season ? styles.dropdownSelected : styles.dropdownPlaceholder}>{season || 'Season of Wear'}</Text>
          <Text>▼</Text>
        </TouchableOpacity>
        {seasonOpen && (
          <View style={styles.dropdownList}>
            {seasons.map(s => (
              <TouchableOpacity key={s} style={styles.dropdownItem} onPress={() => { setSeason(s); setSeasonOpen(false); }}>
                <Text style={styles.dropdownItemText}>{s}</Text>
              </TouchableOpacity>
            ))}
          </View>
        )}
        <View style={styles.row}>
          <TextInput style={[styles.input, { flex: 1, marginRight: 8 }]} placeholder="Shoes Color" value={color} onChangeText={setColor} />
          <TextInput style={[styles.input, { flex: 1 }]} placeholder="Shoes Designer" value={designer} onChangeText={setDesigner} />
        </View>
        <OrangeButton title="Reserve Bin" onPress={() => { if (shoeType) setStep(3); else Alert.alert('Please enter shoe type'); }} />
      </ScrollView>
    </SafeAreaView>
  );

  return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="" onBack={() => navigate('home')} />
      <ScrollView contentContainerStyle={styles.scrollContent}>
        <Text style={styles.cursiveTitle}>Photo Studio</Text>
        <View style={styles.photoStudioBox}>
          <Text style={styles.photoStudioEmoji}>📦</Text>
          <Text style={styles.photoStudioText}>Open the top-center bin, set up the screen and selfie-stick, and place the item to be photographed on the platform, as shown.</Text>
        </View>
        <OrangeButton title="Continue" onPress={() => setStep(2)} />
      </ScrollView>
    </SafeAreaView>
  );
}

function VendShoesScreen({ navigate, shoes, setShoes, currentUser }) {
  const [search, setSearch] = useState('');
  const [selected, setSelected] = useState(null);
  const [step, setStep] = useState(1);
  const [filter, setFilter] = useState('');
  const filtered = shoes.filter(s => s.type.toLowerCase().includes(search.toLowerCase()) || s.color.toLowerCase().includes(search.toLowerCase()) || s.designer.toLowerCase().includes(search.toLowerCase()));

  if (step === 3) return (
    <SafeAreaView style={styles.container}>
      <AppHeader />
      <View style={styles.successContainer}>
        <View style={styles.vendInstructions}>
          <Text style={styles.vendInstructTitle}>Open the indicated bin and remove the selected item, then close the bin.</Text>
          <Text style={styles.vendBinLabel}>{selected?.bin} : {selected?.col} : {selected?.row} : {selected?.location}</Text>
        </View>
        <OrangeButton title="Done" onPress={() => { setShoes(shoes.filter(s => s.id !== selected.id)); navigate('home'); }} />
      </View>
    </SafeAreaView>
  );

  if (step === 2 && selected) return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="Vend Shoes" onBack={() => setStep(1)} />
      <ScrollView contentContainerStyle={styles.scrollContent}>
        <Text style={styles.confirmEmoji}>{selected.emoji}</Text>
        <View style={styles.shoeDetailCard}>
          <DetailRow label="Shoe Type" value={selected.type} />
          <DetailRow label="Construction" value={selected.construction} />
          <DetailRow label="Season of Wear" value={selected.season} />
          <DetailRow label="Shoe Color" value={selected.color} />
          <DetailRow label="Designer" value={selected.designer} />
          <DetailRow label="Location" value={`${selected.bin} : ${selected.col} : ${selected.row}`} />
        </View>
        <OrangeButton title="Vend Shoes" onPress={() => setStep(3)} />
        <View style={{ height: 12 }} />
        <GhostButton title="Permanently Remove" onPress={() => Alert.alert('Permanently Remove', `Remove ${selected.type} permanently?`, [{ text: 'Cancel' }, { text: 'Yes, Remove', style: 'destructive', onPress: () => { setShoes(shoes.filter(s => s.id !== selected.id)); navigate('home'); } }])} />
        <View style={{ height: 12 }} />
        <GhostButton title="Cancel" onPress={() => setStep(1)} />
      </ScrollView>
    </SafeAreaView>
  );

  return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="Vend Shoes" onBack={() => navigate('home')} />
      <View style={styles.searchContainer}>
        <Text style={styles.searchIcon}>🔍</Text>
        <TextInput style={styles.searchInput} placeholder="Search shoe type" value={search} onChangeText={setSearch} />
      </View>
      <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.filterRow}>
        {['Season', 'Color', 'Brand', 'Material'].map(f => (
          <TouchableOpacity key={f} style={[styles.filterChip, filter === f && styles.filterChipActive]} onPress={() => setFilter(filter === f ? '' : f)}>
            <Text style={[styles.filterChipText, filter === f && styles.filterChipTextActive]}>{f}</Text>
          </TouchableOpacity>
        ))}
      </ScrollView>
      <FlatList data={filtered} numColumns={3} keyExtractor={i => i.id} contentContainerStyle={styles.shoeGrid} renderItem={({ item }) => (
        <TouchableOpacity style={styles.shoeThumb} onPress={() => { setSelected(item); setStep(2); }}>
          <Text style={styles.shoeThumbEmoji}>{item.emoji}</Text>
          <Text style={styles.shoeThumbLabel}>{item.type}</Text>
        </TouchableOpacity>
      )} />
    </SafeAreaView>
  );
}

function ReturnShoesScreen({ navigate, shoes, setShoes }) {
  const [search, setSearch] = useState('');
  const [selected, setSelected] = useState(null);
  const filtered = shoes.filter(s => s.type.toLowerCase().includes(search.toLowerCase()));

  if (selected) return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="Return Shoes" onBack={() => setSelected(null)} />
      <ScrollView contentContainerStyle={styles.scrollContent}>
        <Text style={styles.confirmEmoji}>{selected.emoji}</Text>
        <View style={styles.shoeDetailCard}>
          <DetailRow label="Shoe Type" value={selected.type} />
          <DetailRow label="Return to" value={`${selected.bin} : ${selected.col} : ${selected.row}`} />
          <DetailRow label="Season" value={selected.season} />
          <DetailRow label="Color" value={selected.color} />
        </View>
        <OrangeButton title="Return to this Bin" onPress={() => Alert.alert('Success!', `${selected.type} returned to ${selected.bin}`, [{ text: 'OK', onPress: () => navigate('home') }])} />
        <View style={{ height: 12 }} />
        <GhostButton title="Cancel" onPress={() => setSelected(null)} />
      </ScrollView>
    </SafeAreaView>
  );

  return (
    <SafeAreaView style={styles.container}>
      <BackHeader title="Return Shoes" onBack={() => navigate('home')} />
      <View style={styles.searchContainer}>
        <Text style={styles.searchIcon}>🔍</Text>
        <TextInput style={styles.searchInput} placeholder="Search shoe type" value={search} onChangeText={setSearch} />
      </View>
      <FlatList data={filtered} numColumns={3} keyExtractor={i => i.id} contentContainerStyle={styles.shoeGrid} renderItem={({ item }) => (
        <TouchableOpacity style={styles.shoeThumb} onPress={() => setSelected(item)}>
          <Text style={styles.shoeThumbEmoji}>{item.emoji}</Text>
          <Text style={styles.shoeThumbLabel}>{item.type}</Text>
        </TouchableOpacity>
      )} />
    </SafeAreaView>
  );
}
